cmake_minimum_required(VERSION 3.20)
project(sgkz LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CGAL REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(sgkz_core src/gkz_ndim.cpp)
target_include_directories(sgkz_core PUBLIC include)
target_link_libraries(sgkz_core PUBLIC CGAL::CGAL Eigen3::Eigen)
target_compile_options(sgkz_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sgkz src/main.cpp)
target_link_libraries(sgkz PRIVATE sgkz_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(sgkz_tests tests/gkz_ndim_tests.cpp)
  target_link_libraries(sgkz_tests PRIVATE sgkz_core)
  target_compile_definitions(sgkz_tests PRIVATE
    SGKZ_SIX_POINTS_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../examples/six_points.points")
  add_test(NAME sgkz_tests COMMAND sgkz_tests)
endif()
