cmake_minimum_required(VERSION 3.20)
project(gkz_gui_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CGAL REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(gkz_core STATIC ../src/gkz.cpp)
target_include_directories(gkz_core PUBLIC ../include)
target_link_libraries(gkz_core PUBLIC CGAL::CGAL Eigen3::Eigen)
add_library(gkz_gui_adapter STATIC CoreAdapter/GuiSolverAdapter.cpp)
target_include_directories(gkz_gui_adapter PUBLIC CoreAdapter ../include)
target_link_libraries(gkz_gui_adapter PUBLIC gkz_core CGAL::CGAL Eigen3::Eigen)
